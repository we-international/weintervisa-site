  function translate(){
    observer.disconnect();

    while(w.nextNode()){var n=w.currentNode;if(!n.parentElement||n.parentElement.closest('script,style,svg,.wi-lang,[translate=no]'))continue;
      var r=records.get(n);if(!r||n.nodeValue!==r.output)r={source:n.nodeValue};
